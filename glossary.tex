\newglossaryentry{scifi}{
    name={Science Fiction},
    first={Science Fiction (Sci-Fi)},
    text={Sci-Fi},
    plural={Sci-Fi},
    description={A genre of fiction which focuses on the future and advanced technology. Often utilising the generic conventions of space travel, advanced weaponry and technology and logical, deterministic explanations for phenomena.
            \\\cite{yarlagaddafantasy}}
}

\newglossaryentry{fantasy}{
    name=Medieval Fantasy,
    description={Genre of fiction which focuses on an idealised representation of the historcal medieval period. Often featuring the generic conventions of magic, mythical creatures and feudal societies.}
}

\newglossaryentry{todorov}{
    name=Todorovian Narrative,
    description={
            style of narrative structure which features 5 main states that appear in a particular sequence: Establishing equilibrium, event of change, state of disequilibrium, event of repair, improved equilibrium.
            \\\cite{todorov19712}
        }
}

\newglossaryentry{vfx}{
    name=Visual Effects,
    description={A method of representing certain practical effects within a virtual media production. The effects are usually different to a physical object that holds permanence within the scene.}
}

\newglossaryentry{timbre}{
    name=Timbre,
    description={A word used to describe the instruments used to produce a piece of audio. e.g. brass, futuristic, primal.}
}

\newglossaryentry{cooldown}{
    name=Cooldown,
    description={A mechanic in a video game in which the player must wait for a period of time after using an ability before it can be used again.}
}

\newglossaryentry{chaostheory}{
    name=Chaos Theory,
    description={An approach to large systems within multiple scientfic disciplines. It relies on the idea that a large system will have millions of different factors and variables, and a small change in any 1 factor can have large knock on effects causing massive changes in the overall system.}
}

\newglossaryentry{match}{
    name=Match,
    description={A way of separating phases of gameplay into a ``playing'' phase and a ``menu'' phase. The former allowing the player to explore and use all available mechanics within a game and the latter allowing customisation of the player character}
}

\newglossaryentry{playstyle}{
    name=Playstyle,
    description={A way of describing a player's approach to a game, usually utilising key archetypes found in other media texts such as: ``Healer'', ``Tank'', ``Assassin''}
}

\newglossaryentry{respawn}{
    name=Respawn,
    description={Where the player's health is reset to its full amount, and their location is set to a specific location within the map, allowing a player to continue playing after their character is defeated during a \gls{match}}
}
